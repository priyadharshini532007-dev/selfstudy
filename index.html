SJF preemptive / SRTF (sjfp.c)

gcc sjfp.c -o sjfp, ./sjfp

#include<stdio.h>
int main(){
int n,i,k,t=0,d=0,a[20],b[20],r[20],f[20];
float sw=0,st=0;
printf("Processes: ");scanf("%d",&n);
for(i=0;i<n;i++){printf("AT BT P%d: ",i+1);scanf("%d%d",&a[i],&b[i]);r[i]=b[i];}
while(d<n){
k=-1;
for(i=0;i<n;i++)if(a[i]<=t&&r[i]>0&&(k<0||r[i]<r[k]))k=i;
if(k<0){t++;continue;}
r[k]--;t++;
if(r[k]==0){d++;f[k]=t;}}
printf("\nP\tBT\tWT\tTAT\n");
for(i=0;i<n;i++){
int tat=f[i]-a[i];
printf("P%d\t%d\t%d\t%d\n",i+1,b[i],tat-b[i],tat);
sw+=tat-b[i];st+=tat;}
printf("Avg WT=%.2f\nAvg TAT=%.2f\n",sw/n,st/n);
return 0;}

Round Robin (rr.c)

gcc rr.c -o rr, ./rr

#include<stdio.h>
int main(){
int n,q,i,s,t=0,d=0,b[20],r[20],f[20];

printf("\nP\tBT\tWT\tTAT\n");
for(i=0;i<n;i++){
printf("P%d\t%d\t%d\t%d\n",i+1,b[i],f[i]-b[i],f[i]);
sw+=f[i]-b[i];st+=f[i];}
printf("Avg WT=%.2f\nAvg TAT=%.2f\n",sw/n,st/n);

6. Semaphores

Producer consumer with semaphores (pc.c)

gcc pc.c -o pc -pthread, ./pc

#include<stdio.h>
#include<pthread.h>
#include<semaphore.h>
#define N 3
int b[N],in,out;
sem_t e,f,m;
void*prod(void*a){
for(int i=1;i<=6;i++){
sem_wait(&e);sem_wait(&m);
b[in]=i*10;
printf("Produced %d\n",b[in]);
in=(in+1)%N;
sem_post(&m);sem_post(&f);}
return 0;}
void*cons(void*a){
for(int i=1;i<=6;i++){
sem_wait(&f);sem_wait(&m);
printf("Consumed %d\n",b[out]);
out=(out+1)%N;
sem_post(&m);sem_post(&e);}
return 0;}
int main(){
pthread_t p,c;
sem_init(&e,0,N);sem_init(&f,0,0);sem_init(&m,0,1);
pthread_create(&p,0,prod,0);
pthread_create(&c,0,cons,0);
pthread_join(p,0);pthread_join(c,0);
return 0;}

9. Page replacement

FIFO (fifo.c)

gcc fifo.c -o fifo, ./fifo

#include<stdio.h>
int main(){
int n,f,i,j,k=0,h,fl=0,a[50],fr[10];
printf("Pages: ");scanf("%d",&n);
printf("Sequence: ");
for(i=0;i<n;i++)scanf("%d",&a[i]);

for(i=0;i<n;i++){
h=0;
for(j=0;j<f;j++)if(fr[j]==a[i])h=1;
if(!h){fr[k]=a[i];k=(k+1)%f;fl++;}
printf("\n%d : ",a[i]);
for(j=0;j<f;j++){if(fr[j]<0)printf("- ");else printf("%d ",fr[j]);}}

10. Paging

Simulate paging: logical to physical address (paging.c)

gcc paging.c -o paging, ./paging

#include<stdio.h>
int main(){
int n,ps,i,p,o,c=1,pt[50];
printf("Pages: ");scanf("%d",&n);
printf("Page size: ");scanf("%d",&ps);
printf("Frame number for each page (-1 if absent):\n");
for(i=0;i<n;i++){printf("Page %d: ",i);scanf("%d",&pt[i]);}
while(c==1){
printf("\nLogical address (page offset): ");
scanf("%d%d",&p,&o);
if(p>=n||pt[p]<0)printf("Page not in any frame\n");
else printf("Physical: frame %d, offset %d, address %d\n",pt[p],o,pt[p]*ps+o);
printf("Continue (1/0)? ");scanf("%d",&c);}
return 0;}
